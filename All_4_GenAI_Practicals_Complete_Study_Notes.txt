def real_batch(n):
    """Sample n points near a ring of radius ~0.8 centered at the origin."""
    angles = rng.uniform(0, 2 * np.pi, n)
    r = 0.8 + 0.05 * rng.normal(size=n)
    x = np.stack([r * np.cos(angles), r * np.sin(angles)], axis=1)
    return x.astype("float32")

LATENT_DIM = 2
DATA_DIM = 2
HIDDEN =32

def build_generator():
    return keras.Sequential([
        layers.Input(shape=(LATENT_DIM,)),
        layers.Dense(HIDDEN, activation="leaky_relu"),
        layers.Dense(HIDDEN, activation="leaky_relu"),
        layers.Dense(DATA_DIM, activation="tanh"),
    ], name="generator")

def build_discriminator():
    return keras.Sequential([
        layers.Input(shape=(DATA_DIM,)),
        layers.Dense(HIDDEN, activation='leaky_relu'),
        layers.Dense(HIDDEN, activation='leaky_relu'),
        layers.Dense(1, activation='sigmoid'),
    ], name='discriminator')

G = build_generator()
D = build_discriminator()

G.summary()
D.summary()

CELL 8 — Define one training step
WHAT IT DOES: Define one training step.
REMEMBER: GradientTape / ones_like / zeros_like / gradient / apply_gradients

ORIGINAL CODE:
BATCH = 128

@tf.function
def train_step(real):
    # Phase A: train Discriminator
    z = tf.random.normal((BATCH, LATENT_DIM))

    with tf.GradientTape() as d_tape:
        fake = G(z, training=True)
        d_real_pred = D(real, training=True)
        d_fake_pred = D(fake, training=True)

        d_loss = (
            bce(tf.ones_like(d_real_pred), d_real_pred)
            +
            bce(tf.zeros_like(d_fake_pred), d_fake_pred)
        )

    d_grads = d_tape.gradient(
        d_loss,
        D.trainable_variables
    )

    d_optimizer.apply_gradients(
        zip(d_grads, D.trainable_variables)
    )

    # Phase B: train Generator
    z2 = tf.random.normal((BATCH, LATENT_DIM))

    with tf.GradientTape() as g_tape:
        fake2 = G(z2, training=True)
        d_fake2_pred = D(fake2, training=True)

        g_loss = bce(
            tf.ones_like(d_fake2_pred),
            d_fake2_pred
        )

    g_grads = g_tape.gradient(
        g_loss,
        G.trainable_variables
    )

    g_optimizer.apply_gradients(
        zip(g_grads, G.trainable_variables)
    )

    return d_loss, g_loss

print("Training step compiled.")

    d_losses.append(float(d_loss))
    g_losses.append(float(g_loss))

from matplotlib import axes
def show_comparison(original, detected, title, figsize=(15,7)):
  fig, axes = plt.subplots(1,2,figsize=figsize)
  axes[0].imshow(original)
  axes[0].set_title("Original")
  axes[0].axis("off")

  axes[1].imshow(detected)
  axes[1].set_title(title)
  axes[1].axis("off")

  plt.tight_layout()
  plt.show()

def draw_detections(pil_image, boxex, labels, scores, categories, threshold=0.60):
  output = pil_image.copy()
  draw = ImageDraw.Draw(output)
  kept = 0

  for box, label, score in zip(boxex, labels, scores):
    score = float(score)
    if score < threshold:
      continue

    x1,y1,x2,y2 = [float(v) for v in box]
    class_id = int(label)
    class_name = categories[class_id] if 0 <= class_id < len(categories) else str(class_id)
    text = f"{class_name} {score:.2f}"

    draw.rectangle([x1,y1,x2,y2], outline="red", width=3)
    draw.text((x1 + 2, y1 + 2), text, fill="white")
    kept += 1

  return output, kept

device = "cuda" if torch.cuda.is_available() else "cpu"
print("PyTorch:", torch.__version__)
print("Device:", device)
if device == "cuda":
    print("GPU:", torch.cuda.get_device_name(0))

-------------------------------------------------------------------------------------
CELL 2 — Load MusicGen model
WHAT IT DOES: Load MusicGen model.
REMEMBER: from_pretrained / to(device) / eval / sampling_rate

ORIGINAL CODE:
MODEL_ID = "facebook/musicgen-small"
print("Loading processor...")
processor = AutoProcessor.from_pretrained(MODEL_ID)
print("Loading model...")
model = MusicgenForConditionalGeneration.from_pretrained(MODEL_ID)
model = model.to(device)
model.eval()
sample_rate = model.config.audio_encoder.sampling_rate
print("Model loaded:", MODEL_ID)
print("Sampling rate:", sample_rate, "Hz")

-------------------------------------------------------------------------------------
CELL 3 — Process text and generate audio
WHAT IT DOES: Process text and generate audio.
REMEMBER: processor / return_tensors / generate / detach / cpu / numpy

ORIGINAL CODE:
prompt = "calm ambient piano with soft strings, slow tempo, peaceful evening atmosphere"

inputs = processor(text=[prompt], padding=True, return_tensors="pt")
inputs = {key: value.to(device) for key, value in inputs.items()}

torch.manual_seed(42)
if device == "cuda":
    torch.cuda.manual_seed_all(42)

with torch.no_grad():
    audio_values = model.generate(
        **inputs,
        do_sample=True,
        guidance_scale=3.0,
        max_new_tokens=256
    )

audio_musicgen = audio_values[0,0].detach().cpu().numpy()
display(Audio(audio_musicgen, rate=sample_rate))

-------------------------------------------------------------------------------------
CELL 4 — Normalize and save WAV
WHAT IT DOES: Normalize and save WAV.
REMEMBER: abs / max / int16 / write

ORIGINAL CODE:
peak = max(np.max(np.abs(audio_musicgen)), 1e-9)
audio_normalized = audio_musicgen / peak
audio_int16 = np.int16(audio_normalized * 32767)
write("partB_musicgen_output.wav", sample_rate, audio_int16)
print("Saved: partB_musicgen_output.wav")

-------------------------------------------------------------------------------------
CELL 5 — Plot waveform
WHAT IT DOES: Plot waveform.
REMEMBER: arange / sample_rate / plot

ORIGINAL CODE:
time = np.arange(len(audio_musicgen)) / sample_rate
plt.figure(figsize=(12,4))
plt.plot(time, audio_musicgen)
plt.xlabel("Time (seconds)")
plt.ylabel("Amplitude")
plt.title("MusicGen Output - Waveform")

plt.show()

-------------------------------------------------------------------------------------
CELL 6 — Compare two prompts
WHAT IT DOES: Compare two prompts.
REMEMBER: processor / generate / Audio

ORIGINAL CODE:
prompts = [
    "relaxing piano music",
    "gentle solo piano, slow tempo, reflective mood, soft room ambience"
]

for text_prompt in prompts:
    print("Prompt:", text_prompt)
    inputs = processor(text=[text_prompt], padding=True, return_tensors="pt")
    inputs = {key: value.to(device) for key, value in inputs.items()}
    torch.manual_seed(55)
    if device == "cuda":
        torch.cuda.manual_seed_all(55)
    with torch.no_grad():
        output = model.generate(**inputs, do_sample=True, guidance_scale=3.0, max_new_tokens=256)
    clip = output[0,0].detach().cpu().numpy()
    display(Audio(clip, rate=sample_rate))
